// This is ELIZA's ENTIRE intelligence.
// No AI. No neural networks. Just IF/ELSE.

function eliza(input) {

    // Convert to lowercase so "SAD" and "sad" both work
    input = input.toLowerCase();

    // Rule 1: User mentions feeling something
    if (input.includes("i feel")) {
        var emotion = getWordsAfter(input, "i feel");
        return "Why do you feel " + emotion + "?";
    }

    // Rule 2: User mentions their mother
    if (input.includes("mother") || input.includes("mom")) {
        return "Tell me more about your family.";
    }

    // Rule 3: User mentions being sad or depressed
    if (input.includes("sad") || input.includes("depressed")) {
        return "I'm sorry to hear that. What makes you sad?";
    }

    // Rule 4: User says yes or no
    if (input === "yes" || input === "no") {
        return "Can you elaborate on that?";
    }

    // No rule matched? Generic fallback.
    return "Please go on.";
    // ^ This is the trick. When ELIZA doesn't
    //   understand, it pretends to listen.
    //   People thought it was being thoughtful!
}

🎯 The Point of This Demo

  • ELIZA has exactly 4 rules + 1 fallback. That's the entire "AI." No learning, no memory, no understanding.
  • Same input = same output, every time. Try typing "I feel sad" twice. You'll get the exact same response. ChatGPT would give a different answer each time.
  • The fallback is the trick. When ELIZA doesn't understand (which is most of the time), it says "Please go on" -- and people thought it was being a thoughtful therapist!
  • Compare this to ChatGPT: GPT-4 has ~1.8 trillion parameters and was trained on the entire internet. ELIZA has 4 if-statements. That's 60 years of progress.
